pl_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/pl_icon.dds"
	}
	decision_group_type = PM_major_decisions_group
	desc = pl_decision_desc
	selection_tooltip = pl_decision_tooltip

	ai_check_interval = 0

	is_shown = {
		root.promoted_legend.legend_owner ?= root
		is_ai = no
	}

	is_valid_showing_failures_only = {
		is_ai = no
	}

	effect = {
		save_scope_as = actor

		every_living_character = {
			save_scope_as = recipient
			limit = {
				root != this
				in_diplomatic_range = root
				is_landed = yes
				age >= 12
				is_physically_able = yes
				NOT = { government_has_flag = government_is_herder }
				NOT = { exists = promoted_legend }
				
				OR = {
					dynasty = root.dynasty
					knows_language_of_culture = root.culture
					root = {
						has_any_moderate_good_relationship_with_character_trigger = { CHARACTER = this }
					}
					any_character_to_title_neighboring_county = {
						any_county_province = {
							any_province_legend = {
								this = root.promoted_legend
							}
						}
					}
				}

				# Acceptance chance (ai_accept sum, no hook/gold) must be above 0
				pl_acceptance_score > 0
			}
			
			add_to_list = pl_potential_promoters
		}
		
		
		ordered_in_list = {
			list = pl_potential_promoters
			order_by = pl_acceptance_score
			max = 1000
			check_range_bounds = no
			save_scope_as = recipient
			
			custom_tooltip = {
				text = pl_tooltip
			}
		}
	}
	
	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
